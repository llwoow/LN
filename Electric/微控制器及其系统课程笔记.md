## 1. 微控制器基础概念

### 1.1 定义与特点

**微控制器（MCU）**是将CPU、存储器、I/O接口等集成在一个芯片上的微型计算机系统。

#### 主要特点：
- **集成度高**：单芯片集成所有基本功能
- **体积小**：适合嵌入式应用
- **功耗低**：特别适合电池供电应用
- **成本低**：大规模生产，成本优势明显
- **可靠性高**：减少外部连线，降低故障率

### 1.2 分类与选型

#### 按位数分类：
| 类型 | 位数 | 典型应用 | 代表产品 |
|------|------|----------|----------|
| 4位MCU | 4位 | 简单控制 | 早期家电控制器 |
| 8位MCU | 8位 | 基础控制 | 8051、AVR、PIC |
| 16位MCU | 16位 | 中等复杂应用 | MSP430、STM32F0 |
| 32位MCU | 32位 | 复杂应用 | STM32、ESP32 |

#### 按架构分类：
- **冯·诺依曼架构**：程序和数据共享同一存储空间
- **哈佛架构**：程序和数据存储分离，支持并行访问

### 1.3 主要厂商与产品系列

| 厂商 | 主要系列 | 特点 |
|------|----------|------|
| ST | STM32系列 | ARM Cortex-M内核，生态完善 |
| TI | MSP430系列 | 超低功耗，16位RISC |
| Microchip | PIC系列 | 8位/16位/32位，稳定性好 |
| NXP | LPC系列 | ARM内核，工业级应用 |
| 英特尔 | 8051系列 | 经典8位，资料丰富 |
| 瑞萨 | RX系列 | 16/32位，高性能 |

## 2. 微控制器硬件架构

### 2.1 核心处理器（CPU）

#### ARM Cortex-M系列架构特点：

**Cortex-M0/M0+：**
- 3级流水线
- Thumb-2指令集
- 极低功耗
- 成本敏感应用

**Cortex-M3：**
- 3级流水线
- Thumb-2指令集
- 中等性能
- 广泛的应用领域

**Cortex-M4：**
- 3级流水线
- Thumb-2 + DSP指令
- FPU可选
- 数字信号处理

**Cortex-M7：**
- 6级流水线
- 更高性能
- 指令和数据缓存
- 复杂应用

### 2.2 总线系统

#### AHB（Advanced High-performance Bus）
- **特点**：高性能总线
- **用途**：连接CPU、DMA控制器、快速外设
- **特点**：支持突发传输、分离事务

#### APB（Advanced Peripheral Bus）
- **特点**：外设总线
- **用途**：连接低速外设（GPIO、UART等）
- **特点**：功耗低、结构简单

```c
// 电机控制结构体
typedef struct {
    TIM_HandleTypeDef* pwm_timer;     // PWM定时器
    uint32_t pwm_channel;             // PWM通道
    GPIO_TypeDef* dir_port;           // 方向控制端口
    uint16_t dir_pin;                 // 方向控制引脚
    GPIO_TypeDef* enable_port;        // 使能控制端口
    uint16_t enable_pin;              // 使能控制引脚
    uint16_t speed;                   // 速度 (0-1000)
    uint8_t direction;                // 方向 (0=正转, 1=反转)
    uint8_t enabled;                  // 使能状态
} MotorControl_t;

// 编码器数据结构
typedef struct {
    TIM_HandleTypeDef* encoder_timer; // 编码器定时器
    int16_t position;                 // 当前位置
    int16_t last_position;            // 上次位置
    int16_t velocity;                 // 速度 (脉冲/时间)
    uint32_t last_time;               // 上次时间
} Encoder_t;

MotorControl_t motor1, motor2;
Encoder_t encoder1, encoder2;

// 电机初始化
void Motor_Init(MotorControl_t* motor, TIM_HandleTypeDef* pwm_timer, uint32_t pwm_channel,
                GPIO_TypeDef* dir_port, uint16_t dir_pin,
                GPIO_TypeDef* enable_port, uint16_t enable_pin)
{
    motor->pwm_timer = pwm_timer;
    motor->pwm_channel = pwm_channel;
    motor->dir_port = dir_port;
    motor->dir_pin = dir_pin;
    motor->enable_port = enable_port;
    motor->enable_pin = enable_pin;
    motor->speed = 0;
    motor->direction = 0;
    motor->enabled = 0;
    
    // 初始化GPIO
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    
    // 方向控制引脚
    GPIO_InitStruct.Pin = dir_pin;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_HIGH;
    HAL_GPIO_Init(dir_port, &GPIO_InitStruct);
    
    // 使能控制引脚
    GPIO_InitStruct.Pin = enable_pin;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_HIGH;
    HAL_GPIO_Init(enable_port, &GPIO_InitStruct);
    
    // 初始化PWM
    TIM3_PWM_Init();
}

// 电机速度控制
void Motor_SetSpeed(MotorControl_t* motor, uint16_t speed)
{
    if(speed > 1000) speed = 1000;
    
    motor->speed = speed;
    __HAL_TIM_SET_COMPARE(motor->pwm_timer, motor->pwm_channel, speed);
}

// 电机方向控制
void Motor_SetDirection(MotorControl_t* motor, uint8_t direction)
{
    motor->direction = direction;
    HAL_GPIO_WritePin(motor->dir_port, motor->dir_pin, direction);
}

// 电机使能控制
void Motor_Enable(MotorControl_t* motor, uint8_t enable)
{
    motor->enabled = enable;
    HAL_GPIO_WritePin(motor->enable_port, motor->enable_pin, !enable);
}

// 编码器初始化
void Encoder_Init(Encoder_t* encoder, TIM_HandleTypeDef* encoder_timer)
{
    encoder->encoder_timer = encoder_timer;
    encoder->position = 0;
    encoder->last_position = 0;
    encoder->velocity = 0;
    encoder->last_time = HAL_GetTick();
    
    // 启动编码器
    HAL_TIM_Encoder_Start(encoder->encoder_timer, TIM_CHANNEL_1 | TIM_CHANNEL_2);
}

// 编码器位置读取
void Encoder_Update(Encoder_t* encoder)
{
    int16_t current_position = encoder->encoder_timer->Instance->CNT;
    uint32_t current_time = HAL_GetTick();
    uint32_t dt = current_time - encoder->last_time;
    
    // 计算位置变化
    int16_t delta = current_position - encoder->last_position;
    
    // 处理计数器溢出
    if(delta > 32768) delta -= 65536;
    if(delta < -32768) delta += 65536;
    
    // 更新位置
    encoder->position += delta;
    
    // 计算速度 (脉冲/毫秒)
    encoder->velocity = (int16_t)((delta * 1000) / dt);
    
    // 更新状态
    encoder->last_position = current_position;
    encoder->last_time = current_time;
}

// PID电机速度控制
void Motor_SpeedControl(MotorControl_t* motor, Encoder_t* encoder, uint16_t target_speed)
{
    static float integral = 0;
    static int16_t last_error = 0;
    
    // 读取当前速度
    Encoder_Update(encoder);
    int16_t current_speed = encoder->velocity;
    
    // 计算误差
    int16_t error = target_speed - current_speed;
    
    // PID参数
    float kp = 0.1f;
    float ki = 0.01f;
    float kd = 0.05f;
    
    // PID计算
    float p_term = kp * error;
    integral += error;
    float i_term = ki * integral;
    float d_term = kd * (error - last_error);
    
    float output = p_term + i_term + d_term;
    
    // 限制输出范围
    if(output > 500) output = 500;
    if(output < -500) output = -500;
    
    // 设置电机速度
    uint16_t abs_speed = (uint16_t)abs(output);
    Motor_SetSpeed(motor, abs_speed);
    
    // 设置方向
    if(output >= 0)
    {
        Motor_SetDirection(motor, 0);  // 正转
    }
    else
    {
        Motor_SetDirection(motor, 1);  // 反转
    }
    
    last_error = error;
}

// 位置控制
void Motor_PositionControl(MotorControl_t* motor, Encoder_t* encoder, int32_t target_position)
{
    // 读取当前位置
    Encoder_Update(encoder);
    int32_t current_position = encoder->position;
    
    // 计算位置误差
    int32_t position_error = target_position - current_position;
    
    // 位置环PID
    float kp_pos = 0.01f;
    float ki_pos = 0.001f;
    static float pos_integral = 0;
    
    float p_term = kp_pos * position_error;
    pos_integral += position_error;
    float i_term = ki_pos * pos_integral;
    
    float speed_output = p_term + i_term;
    
    // 限制速度输出
    if(speed_output > 1000) speed_output = 1000;
    if(speed_output < -1000) speed_output = -1000;
    
    // 设置电机速度
    uint16_t speed = (uint16_t)abs(speed_output);
    Motor_SetSpeed(motor, speed);
    
    // 设置方向
    if(speed_output >= 0)
    {
        Motor_SetDirection(motor, 0);
    }
    else
    {
        Motor_SetDirection(motor, 1);
    }
}

// 主控制循环
void Motor_Control_Task(void)
{
    // 电机1速度控制
    Motor_SpeedControl(&motor1, &encoder1, 500);  // 目标速度500脉冲/毫秒
    
    // 电机2位置控制
    Motor_PositionControl(&motor2, &encoder2, 10000);  // 目标位置10000脉冲
    
    // 打印状态
    printf("Motor1: Speed=%d, Pos=%d\n", encoder1.velocity, encoder1.position);
    printf("Motor2: Speed=%d, Pos=%d\n", encoder2.velocity, encoder2.position);
}
```

```c
// PLC数据结构
typedef struct {
    uint8_t inputs[16];        // 输入状态
    uint8_t outputs[16];       // 输出状态
    uint8_t coils[32];         // 辅助继电器
    uint16_t timers[8];        // 定时器当前值
    uint8_t timer_done[8];     // 定时器完成标志
    uint16_t counters[8];      // 计数器当前值
    uint8_t counters_done[8];  // 计数器完成标志
} PLC_Status_t;

PLC_Status_t plc_status;

// PLC输入读取
void PLC_ReadInputs(void)
{
    // 读取数字输入
    plc_status.inputs[0] = HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_0);  // 输入0
    plc_status.inputs[1] = HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_1);  // 输入1
    plc_status.inputs[2] = HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_2);  // 输入2
    // ... 其他输入
    
    // 读取模拟输入
    uint16_t adc_value = ADC_SingleConversion();
    float voltage = (float)adc_value * 3.3f / 4095.0f;
    
    // 将模拟输入转换为数字（阈值判断）
    if(voltage > 1.65f)  // 1.65V阈值
        plc_status.inputs[8] = 1;
    else
        plc_status.inputs[8] = 0;
}

// PLC输出写入
void PLC_WriteOutputs(void)
{
    // 写入数字输出
    HAL_GPIO_WritePin(GPIOC, GPIO_PIN_0, plc_status.outputs[0]);  // 输出0
    HAL_GPIO_WritePin(GPIOC, GPIO_PIN_1, plc_status.outputs[1]);  // 输出1
    HAL_GPIO_WritePin(GPIOC, GPIO_PIN_2, plc_status.outputs[2]);  // 输出2
    // ... 其他输出
    
    // PWM输出
    uint16_t pwm_value = plc_status.outputs[8] * 10;  // 0-1000范围
    __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_1, pwm_value);
}

// PLC定时器
void PLC_UpdateTimers(void)
{
    for(int i = 0; i < 8; i++)
    {
        if(plc_status.timer_done[i] == 0)
        {
            plc_status.timers[i] += 1;  // 假设每个扫描周期1ms
            
            // 检查是否达到设定值 (假设设定值为1000ms)
            if(plc_status.timers[i] >= 1000)
            {
                plc_status.timer_done[i] = 1;
                plc_status.coils[16 + i] = 1;  // 定时器触点
            }
        }
    }
}

// PLC计数器
void PLC_UpdateCounters(void)
{
    for(int i = 0; i < 8; i++)
    {
        // 检查计数输入上升沿
        static uint8_t last_inputs[8] = {0};
        
        if(plc_status.inputs[i] && !last_inputs[i])
        {
            if(plc_status.counters[i] < 65535)
            {
                plc_status.counters[i]++;
                
                // 检查是否达到设定值
                if(plc_status.counters[i] >= 10)  // 假设设定值为10
                {
                    plc_status.counters_done[i] = 1;
                    plc_status.coils[24 + i] = 1;  // 计数器触点
                }
            }
        }
        
        last_inputs[i] = plc_status.inputs[i];
    }
}

// PLC梯形图逻辑实现
void PLC_Logic(void)
{
    // 清除输出
    memset(plc_status.outputs, 0, sizeof(plc_status.outputs));
    
    // 逻辑1: 输入0 -> 辅助继电器0
    plc_status.coils[0] = plc_status.inputs[0];
    
    // 逻辑2: 辅助继电器0 AND 输入1 -> 输出0
    plc_status.outputs[0] = plc_status.coils[0] && plc_status.inputs[1];
    
    // 逻辑3: 输入2 OR 辅助继电器1 -> 辅助继电器1
    plc_status.coils[1] = plc_status.inputs[2] || plc_status.coils[1];
    
    // 逻辑4: 定时器T0 -> 输出1
    plc_status.outputs[1] = plc_status.timer_done[0];
    
    // 逻辑5: 计数器C0 -> 输出2
    plc_status.outputs[2] = plc_status.counters_done[0];
    
    // 逻辑6: 保持继电器 (自锁)
    if(plc_status.outputs[3])
    {
        plc_status.coils[2] = 1;
    }
    else if(plc_status.inputs[3])  // 停止条件
    {
        plc_status.coils[2] = 0;
    }
    plc_status.outputs[3] = plc_status.coils[2];
}

// PLC主循环
void PLC_Task(void)
{
    // 1. 读取输入
    PLC_ReadInputs();
    
    // 2. 更新定时器和计数器
    PLC_UpdateTimers();
    PLC_UpdateCounters();
    
    // 3. 执行梯形图逻辑
    PLC_Logic();
    
    // 4. 写入输出
    PLC_WriteOutputs();
    
    // 5. 打印状态（调试用）
    static uint32_t last_print = 0;
    uint32_t current_time = HAL_GetTick();
    
    if(current_time - last_print > 1000)  // 每秒打印一次
    {
        printf("Inputs: ");
        for(int i = 0; i < 8; i++)
        {
            printf("%d ", plc_status.inputs[i]);
        }
        printf("\n");
        
        printf("Outputs: ");
        for(int i = 0; i < 8; i++)
        {
            printf("%d ", plc_status.outputs[i]);
        }
        printf("\n");
        
        last_print = current_time;
    }
}
```
